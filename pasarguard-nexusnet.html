#!/usr/bin/env bash set -euo pipefail REPO_RAW="https://raw.githubusercontent.com/SiNaKeEn/NexusNet-Sub/Themes" RED='\033[0;31m'; GREEN='\033[0;32m'; CYAN='\033[0;36m'; YELLOW='\033[1;33m'; NC='\033[0m' [[ "${EUID}" -eq 0 ]] || { echo -e "${RED}run as root: sudo nexusnetsub${NC}"; exit 1; } dl() { if command -v curl >/dev/null 2>&1; then curl -fsSL "$1" -o "$2" elif command -v wget >/dev/null 2>&1; then wget -q -O "$2" "$1" else echo -e "${RED}curl/wget missing${NC}"; exit 1; fi } clear echo -e "${CYAN}" echo " ========================================" echo " NexusNet Sub Manager" echo " ========================================" echo -e "${NC}" echo " 1) Install / Update 3x-ui theme" echo " 2) Install / Update PasarGuard theme" echo " 3) Remove 3x-ui theme" echo " 4) Remove PasarGuard theme" echo " 5) Status" echo " 6) Exit" echo -n "Choice [1-6]: " read -r c case "$c" in 1) mkdir -p /etc/x-ui/sub echo -e "${YELLOW}Downloading...${NC}" dl "${REPO_RAW}/xui-nexusnet.html" /etc/x-ui/sub/sub.html chmod 644 /etc/x-ui/sub/sub.html echo -e "${GREEN}OK /etc/x-ui/sub/sub.html${NC}" ;; 2) mkdir -p /var/lib/pasarguard/templates/subscription echo -e "${YELLOW}Downloading...${NC}" dl "${REPO_RAW}/pasarguard-nexusnet.html" /var/lib/pasarguard/templates/subscription/index.html chmod 644 /var/lib/pasarguard/templates/subscription/index.html envf="/opt/pasarguard/.env"; mkdir -p "$(dirname "$envf")"; touch "$envf" grep -q '^CUSTOM_TEMPLATES_DIRECTORY=' "$envf" 2>/dev/null \ && sed -i 's|^CUSTOM_TEMPLATES_DIRECTORY=.*|CUSTOM_TEMPLATES_DIRECTORY="/var/lib/pasarguard/templates/"|' "$envf" \ || echo 'CUSTOM_TEMPLATES_DIRECTORY="/var/lib/pasarguard/templates/"' >> "$envf" grep -q '^SUBSCRIPTION_PAGE_TEMPLATE=' "$envf" 2>/dev/null \ && sed -i 's|^SUBSCRIPTION_PAGE_TEMPLATE=.*|SUBSCRIPTION_PAGE_TEMPLATE="subscription/index.html"|' "$envf" \ || echo 'SUBSCRIPTION_PAGE_TEMPLATE="subscription/index.html"' >> "$envf" command -v pasarguard >/dev/null 2>&1 && pasarguard restart || true echo -e "${GREEN}OK PasarGuard theme${NC}" ;; 3) rm -f /etc/x-ui/sub/sub.html echo -e "${GREEN}Removed 3x-ui theme${NC}" ;; 4) rm -f /var/lib/pasarguard/templates/subscription/index.html command -v pasarguard >/dev/null 2>&1 && pasarguard restart || true echo -e "${GREEN}Removed PasarGuard theme${NC}" ;; 5) echo -e "${CYAN}Status${NC}" [[ -f /etc/x-ui/sub/sub.html ]] && echo -e "3x-ui: ${GREEN}installed${NC}" || echo -e "3x-ui: ${YELLOW}no${NC}" [[ -f /var/lib/pasarguard/templates/subscription/index.html ]] && echo -e "PasarGuard: ${GREEN}installed${NC}" || echo -e "PasarGuard: ${YELLOW}no${NC}" ;; 6) exit 0 ;; *) echo -e "${RED}Invalid${NC}"; exit 1 ;; esac
